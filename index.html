SUPABASE REALTIME TABLE SETUP

Supabaseの「SQL Editor」で以下のSQLを実行すると、ランキング・リアルタイム監視・ブロック・イベントが全端末間で同期されます。 

CREATE TABLE IF NOT EXISTS public.echo_rankings (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  mhj_id text UNIQUE NOT NULL,
  time_sec numeric(6,2) NOT NULL,
  stage_reached int DEFAULT 5,
  created_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.mhj_sessions (
  mhj_id text PRIMARY KEY,
  started_at text NOT NULL,
  last_seen text NOT NULL,
  status text NOT NULL,
  is_blocked boolean DEFAULT false
);

CREATE TABLE IF NOT EXISTS public.mhj_events (
  id int PRIMARY KEY DEFAULT 1,
  is_active boolean DEFAULT false,
  title text,
  event_type text,
  started_at text
);

CREATE TABLE IF NOT EXISTS public.mhj_vs_rooms (
  room_id text NOT NULL,
  mhj_id text NOT NULL,
  stage int DEFAULT 1,
  x numeric DEFAULT 50,
  y numeric DEFAULT 250,
  finished boolean DEFAULT false,
  finish_time numeric(6,2) DEFAULT 0,
  jammer_attack_at bigint DEFAULT 0,
  updated_at timestamptz DEFAULT now(),
  PRIMARY KEY (room_id, mhj_id)
);

ALTER TABLE public.echo_rankings ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mhj_sessions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mhj_events ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mhj_vs_rooms ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public all rankings" ON public.echo_rankings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public all sessions" ON public.mhj_sessions FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public all events" ON public.mhj_events FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public all vs rooms" ON public.mhj_vs_rooms FOR ALL USING (true) WITH CHECK (true);
